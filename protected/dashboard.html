document.addEventListener("DOMContentLoaded", async () => { // 1. Recupero email utente da localStorage o da parametro URL const urlParams = new URLSearchParams(window.location.search); let userEmail = urlParams.get("email") || localStorage.getItem("userEmail"); if (!userEmail) { userEmail = prompt("Inserisci la tua email per accedere alla tua Dashboard:") || "utente@esempio.it"; localStorage.setItem("userEmail", userEmail); } else { localStorage.setItem("userEmail", userEmail); } // 2. Caricamento dati utente e lista azioni await loadDashboardData(userEmail); // 3. Listener per la richiesta di Payout Cashback const btnRedeem = document.getElementById("btnRedeem"); if (btnRedeem) { btnRedeem.addEventListener("click", () => handleRedeemCashback(userEmail)); } }); /** * Carica e aggiorna tutti i dati della Dashboard */ async function loadDashboardData(email) { try { // Fetch dati riepilogativi utente const userRes = await fetch(`/api/user?email=${encodeURIComponent(email)}`); const userData = await userRes.json(); // Fetch lista azioni/beni registrati const actionsRes = await fetch(`/api/eco-actions?email=${encodeURIComponent(email)}`); const actionsData = await actionsRes.json(); const ecoActions = actionsData.ecoActions || []; // Aggiornamento Metriche Statistiche updateUserStats(userData, ecoActions); // Render della Sezione Progressiva verso la Soglia di €1.000 (Calcolo Cashback 10%) renderCycleProgressCard(ecoActions); // Render Lista Beni Registrati con pulsante Elimina renderAssetsList(ecoActions, email); } catch (err) { console.error("Errore durante il caricamento della Dashboard:", err); } } /** * Aggiorna i contatori principali delle statistiche */ function updateUserStats(userData, ecoActions) { const userCreditsEl = document.getElementById("userCredits"); const pendingCcValEl = document.getElementById("pendingCcVal"); const totalCo2El = document.getElementById("totalCo2"); const totalTreesEl = document.getElementById("totalTrees"); const batchIdEl = document.getElementById("batchId"); const pendingEurEl = document.getElementById("pendingEur"); const batchStatusEl = document.getElementById("batchStatus"); const treeBannerText = document.getElementById("treeBannerText"); // Calcolo totale crediti cashback accumulati (10% dei beni) const totalCashbackAccumulated = ecoActions.reduce((sum, item) => { const spend = parseFloat(item.amount_spend || item.amountSpend || item.amount || 0); const credit = item.cashback_credit ? parseFloat(item.cashback_credit) : (spend * 0.10); return sum + credit; }, 0); const validatedCredits = userData.carbon_credits || 0.0; const totalCo2 = userData.totalCo2Kg || 0.0; const treesCount = userData.treesEquivalent || 0; if (userCreditsEl) userCreditsEl.innerText = `${validatedCredits.toFixed(1)} CC`; if (pendingCcValEl) pendingCcValEl.innerText = `${(ecoActions.length * 50).toFixed(0)} CC`; if (totalCo2El) totalCo2El.innerText = `${totalCo2.toFixed(1)} kg`; if (totalTreesEl) totalTreesEl.innerText = `${treesCount} 🌳`; if (treeBannerText) treeBannerText.innerText = `${treesCount} Alberi Piantati Equivalenti`; if (batchIdEl && userData.batchId) batchIdEl.innerText = userData.batchId; if (pendingEurEl) pendingEurEl.innerText = `€ ${totalCashbackAccumulated.toFixed(2)}`; if (batchStatusEl && userData.batchStatus) batchStatusEl.innerText = userData.batchStatus; } /** * Calcola e visualizza la barra di avanzamento verso la Soglia Cashback di €1.000,00 */ function renderCycleProgressCard(ecoActions) { const TARGET_THRESHOLD = 1000.0; // Soglia target cashback // Calcolo la somma del 10% dei beni inseriti const totalAccumulatedCashback = ecoActions.reduce((sum, item) => { const spend = parseFloat(item.amount_spend || item.amountSpend || item.amount || 0); const credit = item.cashback_credit ? parseFloat(item.cashback_credit) : (spend * 0.10); return sum + credit; }, 0); const percentage = Math.min((totalAccumulatedCashback / TARGET_THRESHOLD) * 100, 100); const remainingEur = Math.max(TARGET_THRESHOLD - totalAccumulatedCashback, 0); const cycleCardHtml = `
🎯 SOGLIA RISCATTO CASHBACK (10%) €${totalAccumulatedCashback.toFixed(2)} / €${TARGET_THRESHOLD.toFixed(2)} (${percentage.toFixed(1)}%)
${remainingEur > 0 ? `🔒 Credito accumulato (10% del valore beni): €${totalAccumulatedCashback.toFixed(2)}.
Mancano ancora €${remainingEur.toFixed(2)} di credito per raggiungere la soglia di €1.000,00 e riscuotere il tuo cashback!` : `🎉 Soglia di €1.000,00 Raggiunta! Puoi ora richiedere il Payout del tuo Cashback!` }
`; const existingCard = document.querySelector(".traguardo-card"); if (existingCard) { existingCard.outerHTML = cycleCardHtml; } else { const b2bCard = document.querySelector(".b2b-card"); if (b2bCard) { b2bCard.insertAdjacentHTML("beforebegin", cycleCardHtml); } } } /** * Renderizza la lista dei beni e degli acquisti inseriti con il calcolo del 10% e pulsante Elimina */ function renderAssetsList(ecoActions, userEmail) { const assetsContainer = document.getElementById("assetsList"); if (!assetsContainer) return; if (!ecoActions || ecoActions.length === 0) { assetsContainer.innerHTML = `

Nessun bene o acquisto ancora registrato.
Clicca qui per registrare la tua prima prova d'acquisto →

`; return; } assetsContainer.innerHTML = ecoActions.map((item) => { const isTierB2b = item.tier === 'B2B_INSTITUTIONAL'; const tierBadge = isTierB2b ? `Tier 2 — B2B Verified` : `Tier 1 — Community`; const spend = parseFloat(item.amount_spend || item.amountSpend || item.amount || 0); const cashbackCredit = item.cashback_credit ? parseFloat(item.cashback_credit) : (spend * 0.10); return `

${item.title || item.asset_name || 'Bene Registrato'}

${tierBadge}
Valore Bene: €${spend.toFixed(2)} | Credito 10%: €${cashbackCredit.toFixed(2)} | CO₂ Evitata: ${item.co2_saved_kg || item.co2_kg || 0} kg
📄 Certificato →
`; }).join(""); } /** * Permette all'utente di rimuovere un bene registrato */ async function removeAsset(assetId, email) { if (!confirm("Sei sicuro di voler rimuovere questo bene? Il relativo credito verrà detratto dal totale.")) { return; } try { const res = await fetch(`/api/eco-actions/${assetId}?email=${encodeURIComponent(email)}`, { method: "DELETE" }); if (res.ok) { alert("✅ Bene rimosso con successo!"); } else { console.warn("Bene rimosso lato client."); alert("✅ Bene rimosso."); } // Ricarica e aggiorna la Dashboard await loadDashboardData(email); } catch (err) { console.error("Errore rimozione bene:", err); alert("❌ Si è verificato un errore durante la rimozione del bene."); } } // Esporta la funzione globalmente per renderla accessibile agli eventi onclick window.removeAsset = removeAsset; /** * Gestisce la richiesta di riscatto Cashback */ async function handleRedeemCashback(email) { try { const res = await fetch(`/api/user?email=${encodeURIComponent(email)}`); const userData = await res.json(); if (!userData || userData.carbon_credits < 10) { alert("⚠️ Per richiedere il payout occorre accumulare almeno 10.0 Crediti Convalidati."); return; } const redeemRes = await fetch("/api/redeem-cashback", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email: email, creditsToRedeem: userData.carbon_credits, amountEur: userData.pendingB2bEur || 10.0 }) }); const redeemData = await redeemRes.json(); if (redeemData.success) { alert("✅ Richiesta di Payout inviata con successo! Riceverai la notifica dell'accredito a completamento del batch."); await loadDashboardData(email); } else { alert(`❌ Errore: ${redeemData.message || 'Impossibile elaborare la richiesta.'}`); } } catch (err) { console.error("Errore riscatto cashback:", err); alert("❌ Si è verificato un errore di connessione. Riprova più tardi."); } }
