document.addEventListener("DOMContentLoaded", async () => { // 1. Recupero email utente da localStorage o da parametro URL const urlParams = new URLSearchParams(window.location.search); let userEmail = urlParams.get("email") || localStorage.getItem("userEmail"); if (!userEmail) { userEmail = prompt("Inserisci la tua email per accedere alla tua Dashboard:") || "utente@esempio.it"; localStorage.setItem("userEmail", userEmail); } else { localStorage.setItem("userEmail", userEmail); } // 2. Caricamento dati utente e lista azioni await loadDashboardData(userEmail); // 3. Listener per la richiesta di Payout Cashback const btnRedeem = document.getElementById("btnRedeem"); if (btnRedeem) { btnRedeem.addEventListener("click", () => handleRedeemCashback(userEmail)); } }); /** * Carica e aggiorna tutti i dati della Dashboard */ async function loadDashboardData(email) { try { // Fetch dati riepilogativi utente const userRes = await fetch(`/api/user?email=${encodeURIComponent(email)}`); const userData = await userRes.json(); // Fetch lista azioni/beni registrati const actionsRes = await fetch(`/api/eco-actions?email=${encodeURIComponent(email)}`); const actionsData = await actionsRes.json(); const ecoActions = actionsData.ecoActions || []; // Aggiornamento Metriche Statistiche updateUserStats(userData, ecoActions); // Inserimento / Render della Sezione "Traguardo a Cicli Progressivi" renderCycleProgressCard(ecoActions.length, userData.pendingB2bEur || 0); // Render Lista Beni Registrati renderAssetsList(ecoActions); } catch (err) { console.error("Errore durante il caricamento della Dashboard:", err); } } /** * Aggiorna i contatori principali delle statistiche */ function updateUserStats(userData, ecoActions) { const userCreditsEl = document.getElementById("userCredits"); const pendingCcValEl = document.getElementById("pendingCcVal"); const totalCo2El = document.getElementById("totalCo2"); const totalTreesEl = document.getElementById("totalTrees"); const batchIdEl = document.getElementById("batchId"); const pendingEurEl = document.getElementById("pendingEur"); const batchStatusEl = document.getElementById("batchStatus"); const treeBannerText = document.getElementById("treeBannerText"); const validatedCredits = userData.carbon_credits || 0.0; const totalCo2 = userData.totalCo2Kg || 0.0; const treesCount = userData.treesEquivalent || 0; const pendingEur = userData.pendingB2bEur || 0.0; if (userCreditsEl) userCreditsEl.innerText = `${validatedCredits.toFixed(1)} CC`; if (pendingCcValEl) pendingCcValEl.innerText = `${(ecoActions.length * 50).toFixed(0)} CC`; if (totalCo2El) totalCo2El.innerText = `${totalCo2.toFixed(1)} kg`; if (totalTreesEl) totalTreesEl.innerText = `${treesCount} 🌳`; if (treeBannerText) treeBannerText.innerText = `${treesCount} Alberi Piantati Equivalenti`; if (batchIdEl && userData.batchId) batchIdEl.innerText = userData.batchId; if (pendingEurEl) pendingEurEl.innerText = `€ ${pendingEur.toFixed(2)}`; if (batchStatusEl && userData.batchStatus) batchStatusEl.innerText = userData.batchStatus; } /** * Calcola e visualizza la barra di avanzamento a Cicli (Ogni 3 beni = 1 Ciclo / €150) */ function renderCycleProgressCard(totalItems, pendingEur) { const ACTIONS_PER_CYCLE = 3; const VALUE_PER_CYCLE = 150; const currentCycle = Math.floor(totalItems / ACTIONS_PER_CYCLE) + 1; const currentProgress = totalItems % ACTIONS_PER_CYCLE; const actionsInBar = (currentProgress === 0 && totalItems > 0) ? ACTIONS_PER_CYCLE : currentProgress; const percentage = (actionsInBar / ACTIONS_PER_CYCLE) * 100; const completedCycles = Math.floor(totalItems / ACTIONS_PER_CYCLE); const lockedValue = completedCycles * VALUE_PER_CYCLE; const cycleCardHtml = `
🔄 CICLO ${currentCycle} IN CORSO ${actionsInBar} di ${ACTIONS_PER_CYCLE} Beni Registrati
${lockedValue > 0 ? `🔒 €${lockedValue}.00 (${completedCycles} ${completedCycles === 1 ? 'ciclo completato' : 'cicli completati'}) sotto analisi dMRV.
Carica ancora ${ACTIONS_PER_CYCLE - actionsInBar} ${ACTIONS_PER_CYCLE - actionsInBar === 1 ? 'bene' : 'beni'} per sbloccare i prossimi €${VALUE_PER_CYCLE}!` : `Carica ancora ${ACTIONS_PER_CYCLE - actionsInBar} ${ACTIONS_PER_CYCLE - actionsInBar === 1 ? 'bene' : 'beni'} per completare il 1° Traguardo da €${VALUE_PER_CYCLE}.` }
`; // Inserisci la card subito prima della sezione B2B o in testa const existingCard = document.querySelector(".traguardo-card"); if (existingCard) { existingCard.outerHTML = cycleCardHtml; } else { const b2bCard = document.querySelector(".b2b-card"); if (b2bCard) { b2bCard.insertAdjacentHTML("beforebegin", cycleCardHtml); } } } /** * Renderizza la lista dei beni e degli acquisti inseriti */ function renderAssetsList(ecoActions) { const assetsContainer = document.getElementById("assetsList"); if (!assetsContainer) return; if (!ecoActions || ecoActions.length === 0) { assetsContainer.innerHTML = `

Nessun bene o acquisto ancora registrato.
Clicca qui per registrare la tua prima prova d'acquisto →

`; return; } assetsContainer.innerHTML = ecoActions.map((item) => { const isTierB2b = item.tier === 'B2B_INSTITUTIONAL'; const tierBadge = isTierB2b ? `Tier 2 — B2B Verified` : `Tier 1 — Community`; return `

${item.title || 'Bene Registrato'}

${tierBadge}
Categoria: ${item.category || 'Generale'} | CO₂ Evitata: ${item.co2_saved_kg || 0} kg
📄 Certificato dMRV →
`; }).join(""); } /** * Gestisce la richiesta di riscatto Cashback */ async function handleRedeemCashback(email) { try { const res = await fetch(`/api/user?email=${encodeURIComponent(email)}`); const userData = await res.json(); if (!userData || userData.carbon_credits < 10) { alert("⚠️ Per richiedere il payout occorre accumulare almeno 10.0 Crediti Convalidati."); return; } const redeemRes = await fetch("/api/redeem-cashback", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email: email, creditsToRedeem: userData.carbon_credits, amountEur: userData.pendingB2bEur || 10.0 }) }); const redeemData = await redeemRes.json(); if (redeemData.success) { alert("✅ Richiesta di Payout inviata con successo! Riceverai la notifica dell'accredito a completamento del batch."); await loadDashboardData(email); } else { alert(`❌ Errore: ${redeemData.message || 'Impossibile elaborare la richiesta.'}`); } } catch (err) { console.error("Errore riscatto cashback:", err); alert("❌ Si è verificato un errore di connessione. Riprova più tardi."); } }
